\begin{frame}{The problem}
\begin{itemize}
\item We want a bound for $\Pr(|S_n - \mathbb{E} S_n| \ge t)$.
\item The variables are independent and bounded.
\end{itemize}
\begin{block}{Theorem}
$\Pr(|S_n - \mathbb{E} S_n| \ge t) \le 2\exp\left(-2t^2 / \sum_i (b_i - a_i)^2\right)$.
\end{block}
